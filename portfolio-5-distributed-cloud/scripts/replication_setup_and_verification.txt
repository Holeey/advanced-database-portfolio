# Clinix PostgreSQL 18 streaming replication – practical command record
# Windows PowerShell (run administrative commands in an elevated PowerShell)

# Primary postgresql.conf settings (data directory of service postgresql-x64-18):
listen_addresses = '*'
port = 5432
wal_level = replica
max_wal_senders = 10
max_replication_slots = 10

# Replication role (execute in psql connected to the primary as postgres):
CREATE ROLE replicator WITH REPLICATION LOGIN PASSWORD '<set-a-strong-password>';

# Add to primary pg_hba.conf:
host replication replicator 127.0.0.1/32 scram-sha-256

# Create standby base backup (enter replication account password when prompted):
New-Item -ItemType Directory -Force -Path "C:\Postgres\clinix_standby" | Out-Null
& "C:\Program Files\PostgreSQL\18\bin\pg_basebackup.exe" -h 127.0.0.1 -p 5432 -U replicator -D "C:\Postgres\clinix_standby" -Fp -Xs -P -R

# Start standby on port 5433:
& "C:\Program Files\PostgreSQL\18\bin\pg_ctl.exe" -D "C:\Postgres\clinix_standby" -o "-p 5433" -l "C:\Postgres\clinix_standby\standby.log" start

# Verify standby role:
& "C:\Program Files\PostgreSQL\18\bin\psql.exe" -h 127.0.0.1 -p 5433 -U postgres -d clinix_hospital_db -c "SELECT pg_is_in_recovery();"

# Verify streaming connection on primary:
& "C:\Program Files\PostgreSQL\18\bin\psql.exe" -h 127.0.0.1 -p 5432 -U postgres -d clinix_hospital_db -c "SELECT application_name, client_addr, state, sync_state FROM pg_stat_replication;"

# Data propagation test (execute on primary):
CREATE TABLE IF NOT EXISTS replication_test (id SERIAL PRIMARY KEY, message TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
INSERT INTO replication_test (message) VALUES ('Clinix streaming replication test successful');

# Query on standby:
SELECT * FROM replication_test;
